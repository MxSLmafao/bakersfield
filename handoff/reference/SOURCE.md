## `assets/brand/BRAND.md`

````markdown
# Bakersfield brand sheet

## Logo

- Mark: pixel croissant with a dark brown French mustache.
- Source: `assets/brand/logo-mark-source.jpg` (generated with Gemini 3.1 Flash Image via Higgsfield, status: approved).
- Style: soft 16-bit pixel art. Soft pixel edges and gentle gradients. Not harsh 8-bit.

## Palette

| Role | Name | Hex |
|---|---|---|
| Background | Cream | `#F6E7CA` |
| Highlight | Butter | `#FAD488` |
| Croissant | Honey | `#F9B35F` |
| Croissant | Amber | `#EA9346` |
| Croissant | Bake | `#D06C2E` |
| Accent | Crust | `#A8572D` |
| Ink | Mustache | `#462217` |

Colours were taken from the logo pixels, not chosen by eye.

## Typography

- Headings and display: **Pixelify Sans** (Google Fonts, SIL OFL). Large sizes only. Use it for the wordmark, section titles and big numbers.
- Body and UI: **DM Sans** (Google Fonts, SIL OFL). Use it for all running text, buttons and labels.
- Rule: body text is the constraint. Pixel face is for display only.

## Voice

- Warm, plain and proud. Short sentences. No invented facts (menu, prices, hours and location come from the client).

````

## `src/app/page.tsx`

````tsx
import { preload } from 'react-dom';
import { Hero } from '@/components/Hero';
import { StorySection } from '@/components/StorySection';
import { ProgressRail } from '@/components/ProgressRail';
import { Visit } from '@/components/Visit';
import { beats, sequenceBeats } from '@/content/beats';
import { withVersion } from '@/lib/asset-version';

export default function Home() {
  const hero = beats.find((beat) => beat.kind === 'hero');
  const visit = beats.find((beat) => beat.kind === 'outro');

  // First frame of the first story segment, one per screen size. Same URL as the frame loader.
  const first = sequenceBeats[0];
  preload(withVersion(`${first.frames.dir}/0001.${first.frames.ext}`), {
    as: 'image',
    fetchPriority: 'low',
    media: '(min-width: 768px)',
  });
  preload(withVersion(`${first.framesSm.dir}/0001.${first.framesSm.ext}`), {
    as: 'image',
    fetchPriority: 'low',
    media: '(max-width: 767px)',
  });

  return (
    <>
      <main id="main">
        {hero && <Hero beat={hero} />}
        <StorySection />
        {visit && <Visit beat={visit} />}
      </main>
      <ProgressRail />
    </>
  );
}
````

````tsx
'use client';

import { useEffect, useLayoutEffect, useRef, useState, type ReactNode } from 'react';
import styles from './RevealOnView.module.css';

/**
 * Wraps its children in a block that rises and fades in once, the first time
 * it is in view. Later scrolling does not hide it again.
 *
 * The content is visible by default, so it shows without JavaScript. The
 * script hides it (sets data-armed) only when it is below the fold, JavaScript
 * runs and reduced motion is not requested. Without IntersectionObserver it
 * never hides the content.
 */
export function RevealOnView({ className, children }: { className?: string; children: ReactNode }) {
  const ref = useRef<HTMLDivElement>(null);
  const [armed, setArmed] = useState(false);
  const [shown, setShown] = useState(false);

  // Runs before the first paint, so the hidden state never flashes on screen.
  // Only the client sets it, so the server HTML and the first client render match.
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el || !('IntersectionObserver' in window)) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const rect = el.getBoundingClientRect();
    const inView = rect.top < window.innerHeight && rect.bottom > 0;
    if (!inView) setArmed(true);
  }, []);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    if (!('IntersectionObserver' in window)) {
      setShown(true);
      return;
    }
    const observer = new IntersectionObserver(
      (entries) => {
        if (!entries.some((entry) => entry.isIntersecting)) return;
        setShown(true);
        observer.disconnect();
      },
      { threshold: 0.15 },
    );
    observer.observe(el);
    return () => observer.disconnect();
  }, []);

  const cls = className ? `${className} ${styles.reveal}` : styles.reveal;
  return (
    <div
      ref={ref}
      className={cls}
      data-armed={armed ? 'true' : undefined}
      data-shown={shown ? 'true' : 'false'}
    >
      {children}
    </div>
  );
}
````

## `src/components/ScrollSequence.tsx`

````tsx
'use client';

import { useEffect, useImperativeHandle, useRef, useState, type Ref } from 'react';
import type { FrameSet } from '@/content/beats';
import { withVersion } from '@/lib/asset-version';
import { localAt, segmentAt } from '@/lib/segments';
import styles from './ScrollSequence.module.css';

export type ScrollSequenceHandle = {
  /** Set the global progress of the story, 0 to 1. Cheap: no React state. */
  setProgress: (progress: number) => void;
};

export type ScrollSequenceProps = {
  /** Frame sets that play one after the other, as one flat list. */
  segments: readonly FrameSet[];
  /** Still image shown until the first frame is ready. */
  poster: string;
  alt: string;
  /** Parallel frame downloads. */
  concurrency: number;
  /** Cap the canvas backing store at 2x the native frame size. Used for the small set. */
  capToFrame: boolean;
  ref?: Ref<ScrollSequenceHandle>;
};

const LOGO_SRC = withVersion('/brand/logo-mark.webp');
const MAX_DPR = 2;
/** Start loading when the stage is within 1.5 viewports of the screen. */
const LOAD_MARGIN = '150% 0px 150% 0px';
const CAP_FACTOR = 2;

const TODO = 0;
const LOADING = 1;
const DONE = 2;
const FAILED = 3;

const pad = (n: number) => String(n).padStart(4, '0');
const clamp01 = (n: number) => (n < 0 ? 0 : n > 1 ? 1 : n);

/**
 * Draws one frame of a flat list of image sequences into a canvas.
 * Global progress picks the segment, and local progress picks the frame.
 * The frame is fitted with "contain", so the whole square is always visible.
 * The parent StorySection pins the section and calls `setProgress` through the ref.
 *
 * Loading: nothing loads until the stage nears the screen. Then the segment
 * the visitor is in loads first, then the next one, then the rest in order.
 */
export function ScrollSequence({
  segments,
  poster,
  alt,
  concurrency,
  capToFrame,
  ref,
}: ScrollSequenceProps) {
  const stageRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const engineRef = useRef<ScrollSequenceHandle | null>(null);

  const [frameMode, setFrameMode] = useState(false);
  const [posterState, setPosterState] = useState<'pending' | 'ok' | 'failed'>('pending');

  useImperativeHandle(
    ref,
    () => ({
      setProgress: (progress: number) => engineRef.current?.setProgress(progress),
    }),
    [],
  );

  // Check the poster in script, not with onError: a failed image can finish
  // loading before hydration and the event is then missed.
  useEffect(() => {
    const img = new Image();
    img.onload = () => setPosterState('ok');
    img.onerror = () => setPosterState('failed');
    img.src = poster;
    return () => {
      img.onload = null;
      img.onerror = null;
    };
  }, [poster]);

  useEffect(() => {
    const stage = stageRef.current;
    const canvas = canvasRef.current;
    if (!stage || !canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    // Flat frame list: segment offsets and the URL of every frame.
    const offsets: number[] = [];
    const urls: string[] = [];
    for (const set of segments) {
      offsets.push(urls.length);
      for (let i = 0; i < set.count; i++) {
        urls.push(withVersion(`${set.dir}/${pad(i + 1)}.${set.ext}`));
      }
    }
    const total = urls.length;
    if (total === 0) return;
    const segmentCount = segments.length;

    const frameIndexFor = (p: number) => {
      const seg = segmentAt(p, segmentCount);
      const count = segments[seg].count;
      const local = localAt(p, segmentCount, seg);
      return offsets[seg] + Math.max(0, Math.round(local * (count - 1)));
    };

    let cancelled = false;
    let rafId = 0;
    let progress = 0;
    let lastTarget = -1;
    let drawn = -1;
    let active = false;
    let focusSeg = 0;
    let running = 0;
    let nativeSize = 0;
    let shown = false;
    const frames: (HTMLImageElement | null)[] = new Array(total).fill(null);
    const status: Uint8Array = new Uint8Array(total); // all TODO

    const sizeCanvas = () => {
      const rect = stage.getBoundingClientRect();
      const dpr = Math.min(window.devicePixelRatio || 1, MAX_DPR);
      let w = rect.width * dpr;
      let h = rect.height * dpr;
      if (capToFrame && nativeSize > 0) {
        // No point upscaling a small frame into a much larger canvas.
        const k = Math.min(1, (nativeSize * CAP_FACTOR) / Math.max(w, h));
        w *= k;
        h *= k;
      }
      canvas.width = Math.max(1, Math.round(w));
      canvas.height = Math.max(1, Math.round(h));
      // Resizing the canvas resets its context state.
      drawn = -1;
    };

    const draw = () => {
      const target = frameIndexFor(progress);
      // Use the wanted frame, or the nearest loaded one while it loads.
      let src = -1;
      for (let i = target; i >= 0; i--) {
        if (frames[i]) {
          src = i;
          break;
        }
      }
      if (src === -1) {
        for (let i = target + 1; i < total; i++) {
          if (frames[i]) {
            src = i;
            break;
          }
        }
      }
      if (src === -1 || src === drawn) return;
      const img = frames[src];
      if (!img) return;

      const cw = canvas.width;
      const ch = canvas.height;
      const iw = img.naturalWidth;
      const ih = img.naturalHeight;
      // object-fit: contain maths
      const scale = Math.min(cw / iw, ch / ih);
      const dw = iw * scale;
      const dh = ih * scale;
      ctx.imageSmoothingEnabled = true;
      ctx.imageSmoothingQuality = 'high';
      ctx.clearRect(0, 0, cw, ch);
      ctx.drawImage(img, (cw - dw) / 2, (ch - dh) / 2, dw, dh);
      drawn = src;
    };

    const schedule = () => {
      if (rafId !== 0) return;
      rafId = requestAnimationFrame(() => {
        rafId = 0;
        if (!cancelled) draw();
      });
    };

    const loadFrame = (i: number) =>
      new Promise<HTMLImageElement | null>((resolve) => {
        const img = new Image();
        img.decoding = 'async';
        img.onload = () => resolve(img);
        img.onerror = () => resolve(null);
        img.src = urls[i];
      });

    // Next frame to fetch. Only the focus segment and the one after it are
    // wanted. The focus segment goes first, nearest frame first.
    // Other segments load only when the focus moves to them.
    const pickNext = (): number => {
      const target = frameIndexFor(progress);
      for (const seg of [focusSeg, focusSeg + 1]) {
        if (seg >= segmentCount) continue;
        const start = offsets[seg];
        const end = start + segments[seg].count;
        let best = -1;
        let bestDist = Infinity;
        for (let i = start; i < end; i++) {
          if (status[i] !== TODO) continue;
          const dist = seg === focusSeg ? Math.abs(i - target) : i - start;
          if (dist < bestDist) {
            bestDist = dist;
            best = i;
          }
        }
        if (best !== -1) return best;
      }
      return -1;
    };

    const pump = () => {
      if (cancelled || !active) return;
      while (running < concurrency) {
        const i = pickNext();
        if (i < 0) return;
        status[i] = LOADING;
        running++;
        void loadFrame(i).then((img) => {
          running--;
          if (cancelled) return;
          if (img) {
            status[i] = DONE;
            frames[i] = img;
            if (nativeSize === 0) {
              nativeSize = img.naturalWidth;
              sizeCanvas();
            }
            if (!shown) {
              shown = true;
              setFrameMode(true);
            }
            schedule();
          } else {
            status[i] = FAILED;
          }
          pump();
        });
      }
    };

    const setProgress = (p: number) => {
      progress = clamp01(p);
      const seg = segmentAt(progress, segmentCount);
      if (seg !== focusSeg) {
        focusSeg = seg;
        pump();
      }
      const target = frameIndexFor(progress);
      if (target === lastTarget) return;
      lastTarget = target;
      schedule();
    };

    engineRef.current = { setProgress };

    sizeCanvas();
    const ro = new ResizeObserver(() => {
      sizeCanvas();
      schedule();
    });
    ro.observe(stage);

    // Start loading only when the stage is near the screen.
    let io: IntersectionObserver | null = null;
    const start = () => {
      active = true;
      pump();
    };
    if ('IntersectionObserver' in window) {
      io = new IntersectionObserver(
        (entries) => {
          if (!entries.some((entry) => entry.isIntersecting)) return;
          io?.disconnect();
          io = null;
          start();
        },
        { rootMargin: LOAD_MARGIN },
      );
      io.observe(stage);
    } else {
      start();
    }

    return () => {
      cancelled = true;
      io?.disconnect();
      ro.disconnect();
      if (rafId !== 0) cancelAnimationFrame(rafId);
      engineRef.current = null;
      setFrameMode(false);
    };
  }, [segments, concurrency, capToFrame]);

  const showPoster = !frameMode;

  return (
    <div ref={stageRef} className={styles.stage}>
      <canvas
        ref={canvasRef}
        className={styles.canvas}
        role="img"
        aria-label={alt}
        hidden={!frameMode}
      />
      {showPoster &&
        (posterState !== 'failed' ? (
          // eslint-disable-next-line @next/next/no-img-element
          <img className={styles.poster} src={poster} alt={alt} />
        ) : (
          <div className={styles.fallback}>
            {/* Decorative: the section title already names the beat. */}
            {/* eslint-disable-next-line @next/next/no-img-element */}
            <img className={styles.logo} src={LOGO_SRC} alt="" width={1024} height={1024} />
          </div>
        ))}
    </div>
  );
}
````

## `src/components/SmoothScroll.tsx`

````tsx
'use client';

import { useEffect } from 'react';
import Lenis from 'lenis';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { scrollToBeat, setSmoother } from '@/lib/scroll';

gsap.registerPlugin(ScrollTrigger);

const BEAT_LINK = '#beat-';

/**
 * Starts Lenis smooth scrolling and keeps it in step with GSAP ScrollTrigger.
 * Smooth scroll is not started for reduced motion, or on touch screens.
 * Beat links (#beat-<slug>) always go through scrollToBeat, so they land on the
 * right story position. Renders nothing.
 */
export function SmoothScroll() {
  useEffect(() => {
    const onClick = (event: MouseEvent) => {
      if (event.defaultPrevented || event.button !== 0) return;
      if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;
      if (!(event.target instanceof Element)) return;
      const link = event.target.closest<HTMLAnchorElement>(`a[href^="${BEAT_LINK}"]`);
      if (!link) return;
      const slug = (link.getAttribute('href') ?? '').slice(BEAT_LINK.length);
      if (scrollToBeat(slug)) event.preventDefault();
    };
    document.addEventListener('click', onClick);

    const reduce = window.matchMedia('(prefers-reduced-motion: reduce)');
    // Touch screens keep native scrolling and momentum. Lenis is not started there.
    // The scroll helpers then use window.scrollTo with behavior 'smooth'.
    const coarse = window.matchMedia('(pointer: coarse)');
    let lenis: Lenis | null = null;
    let onTick: ((time: number) => void) | null = null;
    let onLenisScroll: (() => void) | null = null;

    if (!reduce.matches && !coarse.matches) {
      lenis = new Lenis({
        duration: 1.1,
        easing: (t: number) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
        wheelMultiplier: 1,
        touchMultiplier: 1.2,
      });
      setSmoother(lenis);

      // Keep ScrollTrigger in step with Lenis.
      onLenisScroll = () => ScrollTrigger.update();
      lenis.on('scroll', onLenisScroll);

      // Drive Lenis from the GSAP ticker so both use one clock.
      onTick = (time: number) => {
        lenis?.raf(time * 1000);
      };
      gsap.ticker.add(onTick);
      gsap.ticker.lagSmoothing(0);
    }

    return () => {
      document.removeEventListener('click', onClick);
      if (lenis) {
        setSmoother(null);
        if (onTick) gsap.ticker.remove(onTick);
        if (onLenisScroll) lenis.off('scroll', onLenisScroll);
        lenis.destroy();
      }
    };
  }, []);

  return null;
}
````

````tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { ScrollSequence, type ScrollSequenceHandle } from './ScrollSequence';
import { sequenceBeats } from '@/content/beats';
import { withVersion } from '@/lib/asset-version';
import { setBeatState, useBeatState } from '@/lib/beat-store';
import { PHONE_QUERY, isCoarsePointer, isPhone, stableViewportHeight, wantsSmallFrames } from '@/lib/device';
import { registerBeatY } from '@/lib/scroll';
import { segmentAt } from '@/lib/segments';
import styles from './StorySection.module.css';

gsap.registerPlugin(ScrollTrigger);
// Mobile browsers resize the viewport when the address bar shows or hides.
// Do not refresh the pin for that. A real width change still refreshes it.
ScrollTrigger.config({ ignoreMobileResize: true });

/** Scroll length per beat, as a multiple of the viewport height. */
const VH_PER_BEAT = 1.3;
/** Shorter scroll per beat on phones, so the story is about five screens. */
const VH_PER_BEAT_PHONE = 0.85;
const SEGMENTS = sequenceBeats.length;
const FRAME_SETS = sequenceBeats.map((beat) => beat.frames);
const SMALL_FRAME_SETS = sequenceBeats.map((beat) => beat.framesSm);
const STORY_POSTER = withVersion(sequenceBeats[0].poster);
const STORY_POSTER_SM = withVersion(sequenceBeats[0].posterSm);
const STORY_ALT = 'A croissant being made, from simple ingredients to a golden bake.';

const pad2 = (n: number) => String(n).padStart(2, '0');

/**
 * Global progress where beat i starts. Aims 0.5% of a segment inside the
 * segment, so rounding never lands on the previous beat.
 */
const beatProgress = (i: number) => (i + 0.005) / SEGMENTS;

/** Pin length in px. Phones use the stable 100svh height, so the address bar does not move it. */
function pinLength(): number {
  const phone = isPhone();
  const vh = isCoarsePointer() ? stableViewportHeight() : window.innerHeight;
  return Math.round(vh * (phone ? VH_PER_BEAT_PHONE : VH_PER_BEAT) * SEGMENTS);
}

/**
 * The poster. The browser picks the small file on phones, before any script runs,
 * so the first paint is the same on the server and in the browser.
 */
function StoryPoster() {
  return (
    <picture className={styles.picture}>
      <source media={PHONE_QUERY} srcSet={STORY_POSTER_SM} />
      <img className={styles.poster} src={STORY_POSTER} alt={STORY_ALT} />
    </picture>
  );
}

/**
 * The six sequence beats as one pinned story. One canvas plays all six
 * frame sets back to back. Each beat has its own copy panel.
 * Reduced motion: no pin and no frames. A plain list with one poster per beat.
 */
export function StorySection() {
  // 'pending' until the motion preference is known, so frames never load in reduced mode.
  const [mode, setMode] = useState<'pending' | 'pinned' | 'reduced'>('pending');
  // Which frame set to use. Chosen once on mount and never switched, so no frames load twice.
  const [frameSet, setFrameSet] = useState<'full' | 'small' | null>(null);
  const sectionRef = useRef<HTMLElement>(null);
  const seqRef = useRef<ScrollSequenceHandle>(null);
  const panelRefs = useRef<(HTMLDivElement | null)[]>([]);
  const { zone, segment } = useBeatState();

  useEffect(() => {
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
    const update = () => setMode(mq.matches ? 'reduced' : 'pinned');
    update();
    mq.addEventListener('change', update);
    return () => mq.removeEventListener('change', update);
  }, []);

  useEffect(() => {
    setFrameSet(wantsSmallFrames() ? 'small' : 'full');
  }, []);

  // Pinned story: one ScrollTrigger drives the frames, the panels and the store.
  useEffect(() => {
    const section = sectionRef.current;
    if (mode !== 'pinned' || !section) return;

    const ctx = gsap.context(() => {
      let lastSegment = -1;
      const unregister: (() => void)[] = [];
      // The panel that is showing now (or fading out). -1 before the first one.
      let shownIndex = -1;

      /**
       * Crossfade between beat panels. Interruptible: every panel that is not
       * the incoming one is either fading out (only the previous one) or
       * hidden at once. So fast scrolling never leaves two panels fully shown
       * or none. Desktop adds a light blur-in. Phones keep the plain rise.
       */
      const showPanel = (next: number) => {
        const incoming = panelRefs.current[next];
        if (!incoming) return;
        const outgoing = shownIndex;
        shownIndex = next;

        panelRefs.current.forEach((el, i) => {
          if (!el || i === next) return;
          gsap.killTweensOf(el);
          if (i === outgoing) {
            // Outgoing: quick fade and a rise of 8px.
            gsap.to(el, { autoAlpha: 0, y: -8, duration: 0.18, ease: 'power1.in' });
          } else {
            gsap.set(el, { autoAlpha: 0, y: 0 });
          }
        });

        const blur = !isPhone();
        gsap.killTweensOf(incoming);
        gsap.fromTo(
          incoming,
          { autoAlpha: 0, y: 14, filter: blur ? 'blur(6px)' : 'none' },
          {
            autoAlpha: 1,
            y: 0,
            filter: blur ? 'blur(0px)' : 'none',
            duration: 0.42,
            // A spring-like settle: a small overshoot, then rest.
            ease: 'back.out(1.2)',
            clearProps: 'filter',
          },
        );
      };

      const apply = (self: ScrollTrigger) => {
        seqRef.current?.setProgress(self.progress);
        const y = window.scrollY;
        if (y < self.start) {
          setBeatState('hero', 0);
          return;
        }
        if (y >= self.end) {
          setBeatState('visit', SEGMENTS - 1);
          return;
        }
        const segment = segmentAt(self.progress, SEGMENTS);
        if (segment !== lastSegment) {
          lastSegment = segment;
          showPanel(segment);
        }
        setBeatState('story', segment);
      };

      const st = ScrollTrigger.create({
        trigger: section,
        start: 'top top',
        end: () => `+=${pinLength()}`,
        pin: true,
        pinSpacing: true,
        anticipatePin: 1,
        invalidateOnRefresh: true,
        onRefresh: apply,
        onUpdate: apply,
        onEnter: apply,
        onEnterBack: apply,
        onLeave: apply,
        onLeaveBack: apply,
      });

      // Each beat is addressable: its anchor is where its segment starts.
      sequenceBeats.forEach((beat, i) => {
        unregister.push(
          registerBeatY(beat.slug, () => st.start + beatProgress(i) * (st.end - st.start)),
        );
      });

      // Open the page at a beat when the URL names one, e.g. #beat-bake.
      const hash = window.location.hash.replace('#beat-', '');
      const hashIndex = sequenceBeats.findIndex((beat) => beat.slug === hash);
      if (hashIndex >= 0) {
        window.scrollTo({ top: st.start + beatProgress(hashIndex) * (st.end - st.start) });
      }

      return () => unregister.forEach((fn) => fn());
    }, section);

    return () => ctx.revert();
  }, [mode]);

  // Reduced motion: mark the beat in view, for the progress rail.
  useEffect(() => {
    if (mode !== 'reduced') return;
    const ids = ['beat-hero', ...sequenceBeats.map((beat) => `beat-${beat.slug}`), 'beat-visit'];
    const targets = ids
      .map((id) => document.getElementById(id))
      .filter((el): el is HTMLElement => el !== null);

    const observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (!entry.isIntersecting) continue;
          const id = entry.target.id;
          if (id === 'beat-hero') setBeatState('hero', 0);
          else if (id === 'beat-visit') setBeatState('visit', SEGMENTS - 1);
          else {
            const i = sequenceBeats.findIndex((beat) => `beat-${beat.slug}` === id);
            if (i >= 0) setBeatState('story', i);
          }
        }
      },
      { rootMargin: '-45% 0px -45% 0px', threshold: 0 },
    );
    targets.forEach((el) => observer.observe(el));
    return () => observer.disconnect();
  }, [mode]);

  const lead = <div className={styles.lead} aria-hidden="true" />;

  if (mode === 'reduced') {
    return (
      <>
        {lead}
        <section id="story" className={styles.list} aria-label="The croissant story">
          <ol className={styles.listInner}>
            {sequenceBeats.map((beat) => (
              <li key={beat.id} id={`beat-${beat.slug}`} className={styles.item}>
                <picture className={styles.picture}>
                  <source media={PHONE_QUERY} srcSet={withVersion(beat.posterSm)} />
                  <img
                    className={styles.still}
                    src={withVersion(beat.poster)}
                    width={720}
                    height={720}
                    loading="lazy"
                    alt=""
                  />
                </picture>
                <div className={styles.card}>
                  <p className={styles.num} aria-hidden="true">
                    {pad2(beat.index)}
                  </p>
                  <h2 id={`beat-${beat.slug}-title`} className={styles.title}>
                    {beat.title}
                  </h2>
                  <p className={styles.body}>{beat.body}</p>
                </div>
              </li>
            ))}
          </ol>
        </section>
      </>
    );
  }

  const small = frameSet === 'small';

  return (
    <>
      {lead}
      <section
        ref={sectionRef}
        id="story"
        className={styles.section}
        aria-label="The croissant story"
      >
        {/* Invisible anchor targets. Clicks are routed to the segment start by scroll.ts. */}
        {sequenceBeats.map((beat) => (
          <span key={beat.id} id={`beat-${beat.slug}`} className={styles.anchor} aria-hidden="true" />
        ))}
        <div className={styles.stage}>
          <div className={styles.media}>
            {mode === 'pinned' && frameSet !== null ? (
              <ScrollSequence
                ref={seqRef}
                segments={small ? SMALL_FRAME_SETS : FRAME_SETS}
                poster={small ? STORY_POSTER_SM : STORY_POSTER}
                alt={STORY_ALT}
                concurrency={small ? 4 : 6}
                capToFrame={small}
              />
            ) : (
              <StoryPoster />
            )}
          </div>
          <div className={styles.copy}>
            <div className={styles.panels}>
              {sequenceBeats.map((beat, i) => (
                <div
                  key={beat.id}
                  ref={(el) => {
                    panelRefs.current[i] = el;
                  }}
                  className={styles.panel}
                >
                  <p className={styles.num} aria-hidden="true">
                    {pad2(beat.index)}
                  </p>
                  <h2 id={`beat-${beat.slug}-title`} className={styles.title}>
                    {beat.title}
                  </h2>
                  <p className={styles.body}>{beat.body}</p>
                </div>
              ))}
            </div>
            <div className={styles.bar} aria-hidden="true">
              {sequenceBeats.map((beat, i) => {
                const state =
                  zone !== 'story' ? 'future' : i === segment ? 'current' : i < segment ? 'past' : 'future';
                return <span key={beat.id} className={styles.cell} data-state={state} />;
              })}
            </div>
          </div>
        </div>
      </section>
    </>
  );
}
````

## `src/components/pov/PovSequence.tsx`

````tsx
'use client';

import { useEffect, useImperativeHandle, useRef, useState, type Ref } from 'react';
import type { FrameSet } from '@/content/beats';
import { withVersion } from '@/lib/asset-version';
import { PHONE_QUERY } from '@/lib/device';
import { segmentAt } from '@/lib/segments';
import styles from './PovSequence.module.css';

/*
 * Copy of the shared ScrollSequence engine with two additions:
 *  1. `fit`: 'cover' fills the stage (centre crop), 'contain' shows the whole frame.
 *  2. `cuts` + `fadeSpan`: segments that start with a hard cut to another shot
 *     are blended on the same canvas. The blend is driven by scroll progress, so
 *     it scrubs both ways. Seamless joins are not listed and never blend.
 * Lazy per-segment loading, nearest-loaded fallback, DPR cap, no idle rAF,
 * the small-set canvas cap and the poster behave as in the original.
 */

export type PovSequenceHandle = {
  /** Set the global progress of the story, 0 to 1. Cheap: no React state. */
  setProgress: (progress: number) => void;
};

export type PovSequenceProps = {
  segments: readonly FrameSet[];
  poster: string;
  posterSm: string;
  alt: string;
  concurrency: number;
  /** Cap the canvas backing store at 2x the native frame size. Used for the small set. */
  capToFrame: boolean;
  fit: 'cover' | 'contain';
  /** Segment indexes that begin with a hard cut. */
  cuts: readonly number[];
  /** Blend length as a fraction of one segment, centred on the cut. */
  fadeSpan: number;
  ref?: Ref<PovSequenceHandle>;
};

const LOGO_SRC = withVersion('/brand/logo-mark.webp');
const MAX_DPR = 2;
/** Start loading when the stage is within 1.5 viewports of the screen. */
const LOAD_MARGIN = '150% 0px 150% 0px';
const CAP_FACTOR = 2;
/** Blend alpha is quantised so tiny scroll moves do not redraw. */
const ALPHA_STEPS = 48;

const TODO = 0;
const LOADING = 1;
const DONE = 2;
const FAILED = 3;

const pad = (n: number) => String(n).padStart(4, '0');
const clamp01 = (n: number) => (n < 0 ? 0 : n > 1 ? 1 : n);
const smooth = (t: number) => t * t * (3 - 2 * t);

export function PovSequence({
  segments,
  poster,
  posterSm,
  alt,
  concurrency,
  capToFrame,
  fit,
  cuts,
  fadeSpan,
  ref,
}: PovSequenceProps) {
  const stageRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const engineRef = useRef<PovSequenceHandle | null>(null);

  const [frameMode, setFrameMode] = useState(false);
  const [posterState, setPosterState] = useState<'pending' | 'ok' | 'failed'>('pending');

  useImperativeHandle(
    ref,
    () => ({
      setProgress: (progress: number) => engineRef.current?.setProgress(progress),
    }),
    [],
  );

  // Check the poster in script, not with onError: a failed image can finish
  // loading before hydration and the event is then missed.
  useEffect(() => {
    const img = new Image();
    img.onload = () => setPosterState('ok');
    img.onerror = () => setPosterState('failed');
    img.src = window.matchMedia(PHONE_QUERY).matches ? posterSm : poster;
    return () => {
      img.onload = null;
      img.onerror = null;
    };
  }, [poster, posterSm]);

  useEffect(() => {
    const stage = stageRef.current;
    const canvas = canvasRef.current;
    if (!stage || !canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const offsets: number[] = [];
    const urls: string[] = [];
    for (const set of segments) {
      offsets.push(urls.length);
      for (let i = 0; i < set.count; i++) {
        urls.push(withVersion(`${set.dir}/${pad(i + 1)}.${set.ext}`));
      }
    }
    const total = urls.length;
    if (total === 0) return;
    const segmentCount = segments.length;
    const cutSet = new Set(cuts.filter((s) => s > 0 && s < segmentCount));
    const half = fadeSpan / 2;

    /** Frame to show for a progress, with the cut windows held on the edge frames. */
    const frameIndexFor = (p: number) => {
      const seg = segmentAt(p, segmentCount);
      const count = segments[seg].count;
      const rel = clamp01(p) * segmentCount - seg;
      const lo = cutSet.has(seg) ? half : 0;
      const hi = cutSet.has(seg + 1) ? 1 - half : 1;
      const local = clamp01((rel - lo) / (hi - lo));
      return offsets[seg] + Math.max(0, Math.round(local * (count - 1)));
    };

    /** Frames to draw: the main one, and an incoming one with its alpha inside a cut window. */
    const resolve = (p: number): { a: number; b: number; alpha: number } => {
      const x = clamp01(p) * segmentCount;
      for (const s of cutSet) {
        const d = x - s;
        if (d > -half && d < half) {
          const raw = (d + half) / (2 * half);
          return {
            a: offsets[s - 1] + segments[s - 1].count - 1,
            b: offsets[s],
            alpha: smooth(raw),
          };
        }
      }
      return { a: frameIndexFor(p), b: -1, alpha: 0 };
    };

    let cancelled = false;
    let rafId = 0;
    let progress = 0;
    let lastKey = '';
    let drawnKey = '';
    let active = false;
    let focusSeg = 0;
    let running = 0;
    let nativeSize = 0;
    let shown = false;
    const frames: (HTMLImageElement | null)[] = new Array(total).fill(null);
    const status: Uint8Array = new Uint8Array(total);

    const sizeCanvas = () => {
      const rect = stage.getBoundingClientRect();
      const dpr = Math.min(window.devicePixelRatio || 1, MAX_DPR);
      let w = rect.width * dpr;
      let h = rect.height * dpr;
      if (capToFrame && nativeSize > 0) {
        const k = Math.min(1, (nativeSize * CAP_FACTOR) / Math.max(w, h));
        w *= k;
        h *= k;
      }
      canvas.width = Math.max(1, Math.round(w));
      canvas.height = Math.max(1, Math.round(h));
      drawnKey = ''; // Resizing resets the canvas.
    };

    /** The wanted frame, or the nearest loaded one while it loads. */
    const loadedNear = (target: number): HTMLImageElement | null => {
      for (let i = target; i >= 0; i--) if (frames[i]) return frames[i];
      for (let i = target + 1; i < total; i++) if (frames[i]) return frames[i];
      return null;
    };

    const paint = (img: HTMLImageElement) => {
      const cw = canvas.width;
      const ch = canvas.height;
      const iw = img.naturalWidth;
      const ih = img.naturalHeight;
      const scale = fit === 'cover' ? Math.max(cw / iw, ch / ih) : Math.min(cw / iw, ch / ih);
      const dw = iw * scale;
      const dh = ih * scale;
      ctx.drawImage(img, (cw - dw) / 2, (ch - dh) / 2, dw, dh);
    };

    const keyFor = (p: number) => {
      const r = resolve(p);
      return { r, key: `${r.a}|${r.b}|${r.b >= 0 ? Math.round(r.alpha * ALPHA_STEPS) : 0}` };
    };

    const draw = () => {
      const { r, key } = keyFor(progress);
      if (key === drawnKey) return;
      const imgA = loadedNear(r.a);
      if (!imgA) return;
      ctx.imageSmoothingEnabled = true;
      ctx.imageSmoothingQuality = 'high';
      ctx.globalAlpha = 1;
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      paint(imgA);
      if (r.b >= 0 && r.alpha > 0.004) {
        const imgB = loadedNear(r.b);
        if (imgB) {
          ctx.globalAlpha = r.alpha;
          paint(imgB);
          ctx.globalAlpha = 1;
        }
      }
      drawnKey = key;
    };

    const schedule = () => {
      if (rafId !== 0) return;
      rafId = requestAnimationFrame(() => {
        rafId = 0;
        if (!cancelled) draw();
      });
    };

    const loadFrame = (i: number) =>
      new Promise<HTMLImageElement | null>((resolve) => {
        const img = new Image();
        img.decoding = 'async';
        img.onload = () => resolve(img);
        img.onerror = () => resolve(null);
        img.src = urls[i];
      });

    // Next frame to fetch. Edge frames of the neighbours come first, so a cut
    // can blend in both directions. Then the focus segment, nearest frame
    // first, then the first frames of the next segment in order.
    const pickNext = (): number => {
      const edges = [focusSeg - 1 >= 0 ? offsets[focusSeg] - 1 : -1, focusSeg + 1 < segmentCount ? offsets[focusSeg + 1] : -1];
      for (const i of edges) if (i >= 0 && status[i] === TODO) return i;
      const target = frameIndexFor(progress);
      for (const seg of [focusSeg, focusSeg + 1]) {
        if (seg >= segmentCount) continue;
        const start = offsets[seg];
        const end = start + segments[seg].count;
        let best = -1;
        let bestDist = Infinity;
        for (let i = start; i < end; i++) {
          if (status[i] !== TODO) continue;
          const dist = seg === focusSeg ? Math.abs(i - target) : i - start;
          if (dist < bestDist) {
            bestDist = dist;
            best = i;
          }
        }
        if (best !== -1) return best;
      }
      return -1;
    };

    const pump = () => {
      if (cancelled || !active) return;
      while (running < concurrency) {
        const i = pickNext();
        if (i < 0) return;
        status[i] = LOADING;
        running++;
        void loadFrame(i).then((img) => {
          running--;
          if (cancelled) return;
          if (img) {
            status[i] = DONE;
            frames[i] = img;
            if (nativeSize === 0) {
              nativeSize = img.naturalWidth;
              sizeCanvas();
            }
            if (!shown) {
              shown = true;
              setFrameMode(true);
            }
            drawnKey = '';
            schedule();
          } else {
            status[i] = FAILED;
          }
          pump();
        });
      }
    };

    const setProgress = (p: number) => {
      progress = clamp01(p);
      const seg = segmentAt(progress, segmentCount);
      if (seg !== focusSeg) {
        focusSeg = seg;
        pump();
      }
      const { key } = keyFor(progress);
      if (key === lastKey) return;
      lastKey = key;
      schedule();
    };

    engineRef.current = { setProgress };

    sizeCanvas();
    const ro = new ResizeObserver(() => {
      sizeCanvas();
      schedule();
    });
    ro.observe(stage);

    let io: IntersectionObserver | null = null;
    const start = () => {
      active = true;
      pump();
    };
    if ('IntersectionObserver' in window) {
      io = new IntersectionObserver(
        (entries) => {
          if (!entries.some((entry) => entry.isIntersecting)) return;
          io?.disconnect();
          io = null;
          start();
        },
        { rootMargin: LOAD_MARGIN },
      );
      io.observe(stage);
    } else {
      start();
    }

    return () => {
      cancelled = true;
      io?.disconnect();
      ro.disconnect();
      if (rafId !== 0) cancelAnimationFrame(rafId);
      engineRef.current = null;
      setFrameMode(false);
    };
  }, [segments, concurrency, capToFrame, fit, cuts, fadeSpan]);

  const showPoster = !frameMode;

  return (
    <div ref={stageRef} className={styles.stage}>
      <canvas ref={canvasRef} className={styles.canvas} role="img" aria-label={alt} hidden={!frameMode} />
      {showPoster &&

            {/* Decorative: the section title already names the beat. */}
            {/* eslint-disable-next-line @next/next/no-img-element */}
            <img className={styles.logo} src={LOGO_SRC} alt="" width={1024} height={1024} />
          </div>
        ))}
    </div>
  );
}
````

````tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { PovSequence, type PovSequenceHandle } from './PovSequence';
import { POV_BEAT_TOTAL, POV_CUT_SEGMENTS, POV_STORY_ALT, povSequenceBeats } from '@/content/pov';
import { withVersion } from '@/lib/asset-version';
import { PHONE_QUERY, isCoarsePointer, isPhone, stableViewportHeight, wantsSmallFrames } from '@/lib/device';
import { registerPovY, scrollToPovBeat, setPovState, usePovState } from '@/lib/pov-store';
import { segmentAt } from '@/lib/segments';
import styles from './PovStory.module.css';

gsap.registerPlugin(ScrollTrigger);
// Mobile browsers resize the viewport when the address bar shows or hides.
// Do not refresh the pin for that. A real width change still refreshes it.
ScrollTrigger.config({ ignoreMobileResize: true });

/** Scroll length per beat, as a multiple of the viewport height. */
const VH_PER_BEAT = 1.3;
const VH_PER_BEAT_PHONE = 0.85;
/** Cross-fade length at a hard cut, as a fraction of one segment. */
const FADE_SPAN = 0.08;
const SEGMENTS = povSequenceBeats.length;
const FRAME_SETS = povSequenceBeats.map((beat) => beat.frames);
const SMALL_FRAME_SETS = povSequenceBeats.map((beat) => beat.framesSm);
const POSTER = withVersion(povSequenceBeats[0].poster);
const POSTER_SM = withVersion(povSequenceBeats[0].posterSm);
const HASH_PREFIX = '#pov-';

const pad2 = (n: number) => String(n).padStart(2, '0');
const beatNumber = (index: number) => `${pad2(index)} / ${pad2(POV_BEAT_TOTAL)}`;

/** Global progress where segment i starts: 0.5% of a segment inside, so rounding never picks the previous one. */
const beatProgress = (i: number) => (i + 0.005) / SEGMENTS;

function pinLength(): number {
  const phone = isPhone();
  const vh = isCoarsePointer() ? stableViewportHeight() : window.innerHeight;
  return Math.round(vh * (phone ? VH_PER_BEAT_PHONE : VH_PER_BEAT) * SEGMENTS);
}

function Progress({ zone, segment }: { zone: string; segment: number }) {
  return (
    <div className={styles.bar} aria-hidden="true">
      {povSequenceBeats.map((beat, i) => {
        const state = zone !== 'story' ? 'future' : i === segment ? 'current' : i < segment ? 'past' : 'future';
        return <span key={beat.id} className={styles.cell} data-state={state} />;
      })}
    </div>
  );
}

/**
 * The six first-person segments as one pinned, scroll-scrubbed story.
 * One canvas plays all six frame sets, full-bleed on landscape screens.
 * On portrait screens the frame sits at the top with the copy below it.
 * Reduced motion: no pin and no frames, a plain list of six stills.
 */
export function PovStory() {
  const [mode, setMode] = useState<'pending' | 'pinned' | 'reduced'>('pending');
  // Chosen once on mount and never switched, so no frames load twice.
  const [frameSet, setFrameSet] = useState<'full' | 'small' | null>(null);
  const sectionRef = useRef<HTMLElement>(null);
  const seqRef = useRef<PovSequenceHandle>(null);
  const panelRefs = useRef<(HTMLDivElement | null)[]>([]);
  const panelsRef = useRef<HTMLDivElement>(null);
  const { zone, segment } = usePovState();

  useEffect(() => {
    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
    const update = () => setMode(mq.matches ? 'reduced' : 'pinned');
    update();
    mq.addEventListener('change', update);
    return () => mq.removeEventListener('change', update);
  }, []);

  useEffect(() => {
    setFrameSet(wantsSmallFrames() ? 'small' : 'full');
  }, []);

  useEffect(() => {
    const section = sectionRef.current;
    if (mode !== 'pinned' || !section) return;

    const ctx = gsap.context(() => {
      let lastSegment = -1;
      let shownPanel = -1;
      const unregister: (() => void)[] = [];

      // One panel in, the previous one out. Anything else is hidden at once,
      // so a fast scroll never leaves two or zero panels behind.
      const showPanel = (next: number) => {
        const els = panelRefs.current;
        const prev = shownPanel;
        shownPanel = next;
        els.forEach((el, i) => {
          if (!el || i === next || i === prev) return;
          gsap.killTweensOf(el);
          gsap.set(el, { autoAlpha: 0 });
        });
        const out = prev >= 0 && prev !== next ? els[prev] : null;
        if (out) {
          gsap.killTweensOf(out);
          gsap.to(out, { autoAlpha: 0, y: -8, duration: 0.16, ease: 'power1.in' });
        }
        const el = els[next];
        if (!el) return;
        // The caption box follows the height of the copy in view.
        if (panelsRef.current) {
          if (prev < 0) gsap.set(panelsRef.current, { height: el.offsetHeight });
          else gsap.to(panelsRef.current, { height: el.offsetHeight, duration: 0.5, ease: 'power3.out', overwrite: true });
        }
        gsap.killTweensOf(el);
        gsap.set(el, { autoAlpha: 0, y: 18 });
        gsap.to(el, { autoAlpha: 1, duration: 0.4, delay: out ? 0.08 : 0, ease: 'power2.out' });
        gsap.to(el, { y: 0, duration: 0.8, delay: out ? 0.08 : 0, ease: 'back.out(1.5)' });
      };

      const apply = (self: ScrollTrigger) => {
        seqRef.current?.setProgress(self.progress);
        const y = window.scrollY;
        if (y < self.start) {
          setPovState('hero', 0);
          return;
        }
        if (y >= self.end) {
          setPovState('outro', SEGMENTS - 1);
          return;
        }
        const seg = segmentAt(self.progress, SEGMENTS);
        if (seg !== lastSegment) {
          lastSegment = seg;
          showPanel(seg);
        }
        setPovState('story', seg);
      };

      // After a resize the copy can wrap differently: fit the caption box again.
      const fitBox = () => {
        const h = shownPanel >= 0 ? panelRefs.current[shownPanel]?.offsetHeight : 0;
        if (h && panelsRef.current) gsap.set(panelsRef.current, { height: h });
      };

      const st = ScrollTrigger.create({
        trigger: section,
        start: 'top top',
        end: () => `+=${pinLength()}`,
        pin: true,
        pinSpacing: true,
        anticipatePin: 1,
        invalidateOnRefresh: true,
        onRefresh: (self) => {
          fitBox();
          apply(self);
        },
        onUpdate: apply,
        onEnter: apply,
        onEnterBack: apply,
        onLeave: apply,
        onLeaveBack: apply,
      });

      const yFor = (i: number) => st.start + beatProgress(i) * (st.end - st.start);
      povSequenceBeats.forEach((beat, i) => {
        unregister.push(registerPovY(beat.slug, () => yFor(i)));
      });

      // Deep links: #pov-<slug> opens the story at the start of that segment.
      const goToHash = () => {
        const slug = window.location.hash.startsWith(HASH_PREFIX)
          ? window.location.hash.slice(HASH_PREFIX.length)
          : '';
        const i = povSequenceBeats.findIndex((beat) => beat.slug === slug);
        if (i >= 0) window.scrollTo({ top: yFor(i) });
      };
      goToHash();
      window.addEventListener('hashchange', goToHash);

      const onClick = (event: MouseEvent) => {
        if (event.defaultPrevented || event.button !== 0) return;
        if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;
        if (!(event.target instanceof Element)) return;
        const link = event.target.closest<HTMLAnchorElement>(`a[href^="${HASH_PREFIX}"]`);
        if (!link) return;
        const slug = (link.getAttribute('href') ?? '').slice(HASH_PREFIX.length);
        if (scrollToPovBeat(slug)) event.preventDefault();
      };
      document.addEventListener('click', onClick);

      return () => {
        unregister.forEach((fn) => fn());
        window.removeEventListener('hashchange', goToHash);
        document.removeEventListener('click', onClick);
      };
    }, section);

    return () => {
      ctx.revert();
      setPovState('hero', 0);
    };
  }, [mode]);

  if (mode === 'reduced') {
    return (
      <section className={styles.list} aria-labelledby="pov-story-title">
        <h2 id="pov-story-title" className={styles.srOnly}>
          How a Bakersfield croissant is made
        </h2>
        <ol className={styles.listInner}>
          {povSequenceBeats.map((beat) => (
            <li key={beat.id} id={`pov-${beat.slug}`} className={styles.item}>
              <picture>
                <source media={PHONE_QUERY} srcSet={withVersion(beat.posterSm)} />
                <img className={styles.still} src={withVersion(beat.poster)} width={960} height={540} loading="lazy" alt="" />
              </picture>
              <div className={styles.card}>
                <p className={styles.num} aria-hidden="true">
                  {beatNumber(beat.index)}
                </p>
                <h3 className={styles.title}>{beat.title}</h3>
                <p className={styles.body}>{beat.body}</p>
              </div>
            </li>
          ))}
        </ol>
      </section>
    );
  }

  const small = frameSet === 'small';

  return (
    <section ref={sectionRef} className={styles.section} aria-labelledby="pov-story-title">
      <h2 id="pov-story-title" className={styles.srOnly}>
        How a Bakersfield croissant is made
      </h2>
      {/* Invisible anchor targets for #pov-<slug>. */}
      {povSequenceBeats.map((beat) => (
        <span key={beat.id} id={`pov-${beat.slug}`} className={styles.anchor} aria-hidden="true" />
      ))}
      <div className={styles.stage}>
        <div className={styles.media}>
          {mode === 'pinned' && frameSet !== null ? (
            <PovSequence
              ref={seqRef}
              segments={small ? SMALL_FRAME_SETS : FRAME_SETS}
              poster={POSTER}
              posterSm={POSTER_SM}
              alt={POV_STORY_ALT}
              concurrency={small ? 4 : 6}
              capToFrame={small}
              fit="cover"
              cuts={POV_CUT_SEGMENTS}
              fadeSpan={FADE_SPAN}
            />
          ) : (
            <picture>
              <source media={PHONE_QUERY} srcSet={POSTER_SM} />
              <img className={styles.poster} src={POSTER} alt={POV_STORY_ALT} width={960} height={540} />
            </picture>
          )}
        </div>
        <div className={styles.copy}>
          <div ref={panelsRef} className={styles.panels}>
            {povSequenceBeats.map((beat, i) => (
              <div
                key={beat.id}
                ref={(el) => {
                  panelRefs.current[i] = el;
                }}
                className={styles.panel}
              >
                <p className={styles.num} aria-hidden="true">
                  {beatNumber(beat.index)}
                </p>
                <h3 className={styles.title}>{beat.title}</h3>
                <p className={styles.body}>{beat.body}</p>
              </div>
            ))}
          </div>
          <Progress zone={zone} segment={segment} />

            <button
              key={beat.id}
              type="button"
              className={styles.railButton}

            </button>
          ))}

  );
}
````

## `src/lib/device.ts`

````ts
/**
 * Phones: narrow portrait screens, and short landscape screens.
 * Desktop windows do not match, so their behaviour stays as it was.
 */
export const PHONE_QUERY = '(max-width: 767px), (max-width: 932px) and (max-height: 500px)';

const SLOW_CONNECTIONS: readonly string[] = ['slow-2g', '2g', '3g'];

/** The non-standard Network Information API, typed without `any`. */
type NetworkInformationLike = { saveData?: boolean; effectiveType?: string };
type NavigatorWithConnection = Navigator & { connection?: NetworkInformationLike };

export function isPhone(): boolean {
  return window.matchMedia(PHONE_QUERY).matches;
}

export function isCoarsePointer(): boolean {
  return window.matchMedia('(pointer: coarse)').matches;
}

/**
 * True when the small frame set should be used: a phone, a data saver
 * visitor, or a slow connection. Call once, on mount.
 */
export function wantsSmallFrames(): boolean {
  if (isPhone()) return true;
  const connection = (navigator as NavigatorWithConnection).connection;
  if (!connection) return false;
  if (connection.saveData === true) return true;
  return typeof connection.effectiveType === 'string' && SLOW_CONNECTIONS.includes(connection.effectiveType);
}

/**
 * Height of 100svh, the smallest viewport. It does not change when the
 * mobile address bar shows or hides. Falls back to innerHeight.
 */
export function stableViewportHeight(): number {
  const probe = document.createElement('div');
  probe.setAttribute('aria-hidden', 'true');
  probe.style.cssText =
    'position:fixed;top:0;left:0;width:0;height:100vh;height:100svh;visibility:hidden;pointer-events:none';
  document.body.appendChild(probe);
  const height = probe.getBoundingClientRect().height;
  probe.remove();
  return height > 0 ? height : window.innerHeight;
}
````

````ts
import type Lenis from 'lenis';

let smoother: Lenis | null = null;
const pinnedBeatY = new Map<string, () => number>();

/** Called by SmoothScroll when Lenis starts or stops. */
export function setSmoother(instance: Lenis | null): void {
  smoother = instance;
}

/**
 * The pinned story registers the scroll position where each beat starts.
 * Returns a function that removes the registration.
 */
export function registerBeatY(slug: string, getY: () => number): () => void {
  pinnedBeatY.set(slug, getY);
  return () => {
    if (pinnedBeatY.get(slug) === getY) pinnedBeatY.delete(slug);
  };
}

export function scrollToY(y: number): void {
  if (smoother) {
    smoother.scrollTo(y, { duration: 1.2 });
    return;
  }
  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  window.scrollTo({ top: y, behavior: reduce ? 'auto' : 'smooth' });
}

/**
 * Scrolls to the beat with this slug. Uses the pinned story position when
 * one is registered, else the element with id `beat-<slug>`.
 * Returns false when the beat cannot be found.
 */
export function scrollToBeat(slug: string): boolean {
  const getY = pinnedBeatY.get(slug);
  if (getY) {
    scrollToY(getY());
    return true;
  }
  const el = document.getElementById(`beat-${slug}`);
  if (!el) return false;
  scrollToY(el.getBoundingClientRect().top + window.scrollY);
  return true;
}
````

## `src/lib/segments.ts`

````ts
const clamp01 = (n: number) => (n < 0 ? 0 : n > 1 ? 1 : n);

/** Segment index for a global progress 0..1, split into `count` equal parts. */
export function segmentAt(progress: number, count: number): number {
  if (count <= 0) return 0;
  // The small epsilon stops a boundary value like 4/6 from rounding down to the previous segment.
  return Math.min(count - 1, Math.floor(clamp01(progress) * count + 1e-9));
}

/** Progress inside one segment, 0..1. */
export function localAt(progress: number, count: number, segment: number): number {
  return clamp01(clamp01(progress) * count - segment);
}
````
